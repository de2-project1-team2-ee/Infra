# 1. 인그레스(ALB) 삭제 (이게 지워져야 AWS 로드밸런서가 사라져)
kubectl delete ingress eunbee-ingress -n dev

# 2. 서비스와 디플로이먼트 삭제
kubectl delete svc eunbee-svc -n dev
kubectl delete deployment eunbee-app -n dev

# 3. Helm으로 설치한 컨트롤러 삭제
helm uninstall aws-load-balancer-controller -n kube-system

# 5. iamserviceaccount 삭제 (이름이 겹쳤던 그 설정을 지우는 거야)
eksctl delete iamserviceaccount \
  --cluster=$CLUSTER_NAME \
  --namespace=kube-system \
  --name=aws-load-balancer-controller \
  --wait

# 6. IAM 역할(Role)이 남았는지 확인
aws iam list-roles | grep AmazonEKS-LB-Controller-Role


############################
# 인그레스(ALB) 확인 - "No resources found"가 나오면 성공!
kubectl get ingress -n dev

# 서비스 및 디플로이먼트 확인 - 이것도 비어있어야 해
kubectl get svc,deploy -n dev
