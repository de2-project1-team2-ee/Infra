cat << 'EOF' > 10_install_kubecost.sh
#!/bin/bash

# 1. 환경 변수 로드 (싱가포르 리전 및 계정 정보 로드)
source ./env_config.sh || exit 1

echo "--------------------------------------------------------"
echo "🛠️  [Step 1] EBS CSI Driver Setup (Team-${TEAM_NUMBER})"
echo "--------------------------------------------------------"

# 1-1. IAM OIDC Provider 활성화
echo ">> Associating IAM OIDC Provider..."
eksctl utils associate-iam-oidc-provider \
  --cluster $CLUSTER_NAME \
  --region $AWS_REGION \
  --approve

# 1-2. IAM ServiceAccount 생성
echo ">> Creating IAM Service Account for EBS CSI..."
eksctl create iamserviceaccount \
  --name ebs-csi-controller-sa \
  --namespace kube-system \
  --cluster $CLUSTER_NAME \
  --region $AWS_REGION \
  --role-name "AmazonEKS-EBS-CSI-DriverRole-team-${TEAM_NUMBER}" \
  --role-only \
  --attach-policy-arn arn:aws:iam::aws:policy/service-role/AmazonEBSCSIDriverPolicy \
  --approve --override-existing-serviceaccounts

# 1-3. EBS CSI Driver 애드온 설치 또는 업데이트 (에러 해결 로직)
echo ">> Installing/Updating EBS CSI Driver Add-on..."
aws eks create-addon \
  --cluster-name $CLUSTER_NAME \
  --region $AWS_REGION \
  --addon-name aws-ebs-csi-driver \
  --service-account-role-arn "arn:aws:iam::${AWS_ACCOUNT_ID}:role/AmazonEKS-EBS-CSI-DriverRole-team-${TEAM_NUMBER}" 2>/dev/null \
|| aws eks update-addon \
  --cluster-name $CLUSTER_NAME \
  --region $AWS_REGION \
  --addon-name aws-ebs-csi-driver \
  --service-account-role-arn "arn:aws:iam::${AWS_ACCOUNT_ID}:role/AmazonEKS-EBS-CSI-DriverRole-team-${TEAM_NUMBER}"

# 1-4. 드라이버가 완전히 올라올 때까지 잠시 대기
echo "⏳ Waiting for EBS CSI Driver to initialize..."
sleep 30 

echo "--------------------------------------------------------"
echo "📊 [Step 2] Kubecost Installation (v2.8.6)"
echo "--------------------------------------------------------"

# 2-1. Namespace 생성
kubectl create namespace kubecost --dry-run=client -o yaml | kubectl apply -f -

# 2-2. Helm Repo 업데이트
helm repo add kubecost https://kubecost.github.io/cost-analyzer/
helm repo update

# 2-3. Kubecost 설치 (gp2 스토리지 명시)
echo ">> Installing Kubecost via Helm..."
helm upgrade --install kubecost kubecost/cost-analyzer \
  --namespace kubecost \
  --version 2.8.6 \
  --set kubecostToken="free" \
  --set global.clusterId="${CLUSTER_NAME}" \
  --set persistentVolume.storageClass=gp2 \
  --set prometheus.server.persistentVolume.storageClass=gp2

# 2-4. Kubecost Ingress 생성 (dev 네임스페이스와 연동)
echo ">> Creating Kubecost Ingress for External Access..."
cat <<EOT > kubecost-ingress.yaml
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: kubecost-ingress
  namespace: kubecost
  annotations:
    alb.ingress.kubernetes.io/scheme: internet-facing
    alb.ingress.kubernetes.io/target-type: ip
    alb.ingress.kubernetes.io/load-balancer-name: "k8s-kubecost-t${TEAM_NUMBER}"
spec:
  ingressClassName: alb
  rules:
    - http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: kubecost-cost-analyzer
                port:
                  number: 9090
EOT

kubectl apply -f kubecost-ingress.yaml

echo "--------------------------------------------------------"
echo "✅ team-${TEAM_NUMBER} Kubecost 배포 완료!"
echo "💡 확인: kubectl get pods -n kubecost"
echo "💡 주소확인: kubectl get ingress -n kubecost(2~3분 소요)"
echo "--------------------------------------------------------"
EOF

chmod +x 10_install_kubecost.sh
./10_install_kubecost.sh
